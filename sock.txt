// server.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/socket.h>
#include <netinet/in.h>

#define PORT 5000

// Function to reverse a string
void reverseString(char *str) {
    int len = strlen(str);
    for (int i = 0; i < len/2; i++) {
        char temp = str[i];
        str[i] = str[len - i - 1];
        str[len - i - 1] = temp;
    }
}

int main() {
    int serverSocket, clientSocket;
    struct sockaddr_in server, client;
    socklen_t c = sizeof(client);
    char buffer[1024];

    // 1. Create socket
    serverSocket = socket(AF_INET, SOCK_STREAM, 0);
    if (serverSocket < 0) {
        perror("Socket creation failed");
        exit(1);
    }

    // 2. Setup server address
    server.sin_family = AF_INET;
    server.sin_addr.s_addr = INADDR_ANY;
    server.sin_port = htons(PORT);

    // 3. Bind
    if (bind(serverSocket, (struct sockaddr*)&server, sizeof(server)) < 0) {
        perror("Bind failed");
        exit(1);
    }

    // 4. Listen
    listen(serverSocket, 5);
    printf("Server listening on port %d...\n", PORT);

    // 5. Accept client
    clientSocket = accept(serverSocket, (struct sockaddr*)&client, &c);
    if (clientSocket < 0) {
        perror("Accept failed");
        exit(1);
    }
    printf("Client connected!\n");

    // 6. Receive string
    memset(buffer, 0, sizeof(buffer));
    recv(clientSocket, buffer, sizeof(buffer), 0);
    printf("Received: %s\n", buffer);

    // Reverse the string
    reverseString(buffer);

    // 7. Send reversed string back
    send(clientSocket, buffer, strlen(buffer), 0);
    printf("Sent reversed string: %s\n", buffer);

    // 8. Close sockets
    close(clientSocket);
    close(serverSocket);

    return 0;
}
                  



// client.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/socket.h>
#include <arpa/inet.h>

#define PORT 5000

int main() {
    int sock;
    struct sockaddr_in server;
    char buffer[1024], result[1024];

    // 1. Create socket
    sock = socket(AF_INET, SOCK_STREAM, 0);
    if (sock < 0) {
        perror("Socket creation failed");
        exit(1);
    }

    // 2. Setup server address
    server.sin_family = AF_INET;
    server.sin_addr.s_addr = inet_addr("127.0.0.1"); // localhost
    server.sin_port = htons(PORT);

    // 3. Connect to server
    if (connect(sock, (struct sockaddr*)&server, sizeof(server)) < 0) {
        perror("Connection failed");
        exit(1);
    }
    printf("Connected to server\n");

    // 4. Input string
    printf("Enter a string: ");
    fgets(buffer, sizeof(buffer), stdin);
    buffer[strcspn(buffer, "\n")] = 0; // remove newline

    // 5. Send string
    send(sock, buffer, strlen(buffer), 0);

    // 6. Receive reversed string
    memset(result, 0, sizeof(result));
    recv(sock, result, sizeof(result), 0);
    printf("Reversed string from server: %s\n", result);

    // 7. Close socket
    close(sock);

    return 0;
}
